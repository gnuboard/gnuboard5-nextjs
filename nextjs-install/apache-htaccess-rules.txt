#### G5 Next.js 25 static theme rewrite BEGIN ####
# Put the mobile canonical rules immediately after
# "RewriteEngine On" / "RewriteBase /" and before any existing file/dir
# passthrough condition. /mobile and /mobile/shop are real directories, so Apache's
# DirectorySlash handling otherwise wins before the bridge can redirect it.
# Put the remaining bridge rules after the file/dir passthrough condition and
# before the legacy Gnuboard short-url rules.
#
# The root .htaccess file is commonly environment-specific and may be ignored
# by git, so this file is the tracked deployable copy for the nextjs_default theme.

<IfModule mod_mime.c>
AddType application/manifest+json .webmanifest
</IfModule>

<IfModule mod_alias.c>
RedirectMatch 301 ^/mobile$ /
RedirectMatch 301 ^/mobile/shop/?$ /shop
RedirectMatch 301 ^/shop/$ /shop
</IfModule>

# Home. The root is a real directory, so the passthrough would send it straight to
# Gnuboard's index.php; through the bridge it gets the session guard
# (plugin/webapp/session_guard.php — no session-folder scan stalls on shared hosting).
RewriteRule ^$ theme/nextjs_default/route.php [QSA,L]

# Mobile Gnuboard home routes handled by the Next.js 25 bridge before the real
# /mobile directory or file can win.
RewriteRule ^mobile/?$ theme/nextjs_default/route.php [QSA,L]
RewriteRule ^mobile/index\.php$ theme/nextjs_default/route.php [QSA,L]
RewriteRule ^mobile/(content|group)\.php$ theme/nextjs_default/route.php [QSA,L]

# Public Gnuboard BBS entrypoints handled by the Next.js 25 bridge before the
# original PHP files can win.
RewriteRule ^bbs/(login|register|register_form|register_result|password_lost|poll_result|qalist|qaview|qawrite|memo|memo_form|memo_view|profile|point|scrap|board|write|content|group|faq|new|search)\.php$ theme/nextjs_default/route.php [QSA,L]

# Mobile YoungCart canonical pages handled by the Next.js 25 bridge. Keep
# payment callback subdirectories such as /mobile/shop/toss/* on original PHP.
RewriteRule ^mobile/shop/?$ theme/nextjs_default/route.php [QSA,L]
RewriteRule ^mobile/shop/(index|cart|wishlist|category|coupon|event|item|iteminfo|itemqa|itemqaform|itemrecommend|itemstocksms|itemuse|itemuseform|largeimage|list|listtype|mypage|orderaddress|orderform|orderinquiry|orderinquiryview|personalpay|personalpayform|personalpayresult|search)\.php$ theme/nextjs_default/route.php [QSA,L]

# Canonicalized YoungCart PHP entrypoints that should redirect or render through
# the Next.js 25 static bridge before the original PHP file wins.
RewriteRule ^shop/(index|cart|wishlist|couponzone|search|largeimage|mypage|orderinquiry|personalpay|category|event|item|iteminfo|list|listtype|orderform|orderinquirycancel|orderinquiryview|personalpayform|personalpayresult)\.php$ theme/nextjs_default/route.php [QSA,L]

# Public metadata files should remain available from the site root while the
# nextjs_default theme is active. The PHP bridge avoids leaking this theme's PWA
# metadata after an administrator switches to another theme.
RewriteRule ^(robots\.txt|sitemap(?:-posts)?\.xml|manifest\.webmanifest|manifest\.json|sw\.js|favicon\.ico|icon-[0-9]+\.png|og-default\.png|kcp-bridge\.html)$ theme/nextjs_default/route.php?g5_nextjs_default_asset=$1 [QSA,L]

# Next.js runtime chunks and font files can be requested from /_next/* after
# client-side navigation even though the static export is stored below the
# selected theme directory.
RewriteRule ^_next/(.*)$ theme/nextjs_default/app/_next/$1 [QSA,L]

# Static export RSC payloads are requested as /route.txt?_rsc=... by the
# Next.js client. The PHP route bridge maps dynamic routes to their sentinel
# export payloads, for example boards/free.txt -> boards/__g5_static__.txt.
RewriteRule ^(.+)\.txt$ theme/nextjs_default/route.php [QSA,L]

# RSS remains a PHP feed because Gnuboard already owns the XML output.
RewriteRule ^boards/([0-9a-zA-Z_]+)/rss$ theme/nextjs_default/route.php [QSA,L]
RewriteRule ^rss/([0-9a-zA-Z_]+)$ bbs/rss.php?bo_table=$1 [QSA,L]

# Canonical Next.js routes that do not have legacy PHP entry files.
RewriteRule ^(admin|boards|content|members|mypage)(/.*)?$ theme/nextjs_default/route.php [QSA,L]
RewriteRule ^(faq|forgot-password|login|offline|polls|recent|register|search)$ theme/nextjs_default/route.php [QSA,L]
RewriteRule ^shop/(cart|categories|compare|content|couponzone|events|largeimage|login|order|orders|payment|personalpay|products|qas|register|reviews|search|wishlist)(/.*)?$ theme/nextjs_default/route.php [QSA,L]

# Legacy Gnuboard/YoungCart short URLs should also enter the Next app shell
# directly, while links shown by the theme stay in the legacy short-url shape.
RewriteRule ^shop/list-([0-9a-z]+)$ theme/nextjs_default/route.php [QSA,L]
RewriteRule ^shop/type-([0-9a-z]+)$ theme/nextjs_default/route.php [QSA,L]
RewriteRule ^shop/([0-9a-zA-Z_\-]+)$ theme/nextjs_default/route.php [QSA,L]
RewriteRule ^shop/([^/]+)/$ theme/nextjs_default/route.php [QSA,L]
RewriteRule ^([0-9a-zA-Z_]+)$ theme/nextjs_default/route.php [QSA,L]
RewriteRule ^([0-9a-zA-Z_]+)/([^/]+)/$ theme/nextjs_default/route.php [QSA,L]
RewriteRule ^([0-9a-zA-Z_]+)/write$ theme/nextjs_default/route.php [QSA,L]
RewriteRule ^([0-9a-zA-Z_]+)/([0-9]+)$ theme/nextjs_default/route.php [QSA,L]
#### G5 Next.js 25 static theme rewrite END ####

# 제목 주소(cf_bbs_rewrite=2)는 링크에 끝 슬래시가 붙지 않는다. Next 가 렌더링할 때
# trailingSlash=false 규칙에 따라 떼어 내기 때문이다. 슬래시 있는 형태만 받으면
# 글을 눌렀을 때 404 가 난다. 브리지는 두 형태를 모두 풀 수 있으므로 여기서 함께 넘긴다.
RewriteRule ^(?!admin/|adm/|api/|bbs/|content/|css/|data/|img/|install/|js/|lib/|members/|mobile/|plugin/|rss/|shop/|theme/)([0-9a-zA-Z_]+)/([^/]+)$ theme/nextjs_default/route.php [QSA,L]
